import java.math.BigInteger;

/**
 * A CPL {@code Int}: an arbitrary-precision integer, held as a {@code long} while the value fits.
 * Kotlin's operators apply: {@code a + b}, {@code a - b}, {@code a * b}, {@code -a}, {@code a < b}.
 */
public final class Int implements java.lang.Comparable<Int>, Value {
    private final long s;
    /** Non-null exactly when the value does not fit in a long. */
    private final BigInteger b;

    private static final Int[] CACHE = new Int[1152];

    static {
        for (int i = 0; i < CACHE.length; i++) {
            CACHE[i] = new Int(i - 128, null);
        }
    }

    private Int(long s, BigInteger b) {
        this.s = s;
        this.b = b;
    }

    public static Int of(long v) {
        if (v >= -128 && v < 1024) {
            return CACHE[(int) v + 128];
        }
        return new Int(v, null);
    }

    public static Int of(BigInteger v) {
        if (v.bitLength() < 64) {
            return of(v.longValue());
        }
        return new Int(0, v);
    }

    /** An integer literal of the program. */
    public static Int lit(java.lang.String t) {
        Int v = parse(t);
        if (v == null) {
            throw new java.lang.IllegalArgumentException(t);
        }
        return v;
    }

    /** The decimal grammar of parse_int, {@code ("+" | "-")? ASCII_DIGIT+}; null when t is not one. */
    public static Int parse(java.lang.String t) {
        int i = 0;
        int n = t.length();
        if (n > 0 && (t.charAt(0) == '+' || t.charAt(0) == '-')) {
            i = 1;
        }
        if (i == n) {
            return null;
        }
        for (int k = i; k < n; k++) {
            char c = t.charAt(k);
            if (c < '0' || c > '9') {
                return null;
            }
        }
        if (n - i <= 18) {
            return of(java.lang.Long.parseLong(t));
        }
        return of(new BigInteger(t));
    }

    public boolean fitsLong() {
        return b == null;
    }

    /** The value as a long; an ArithmeticException when it does not fit. */
    public long toLong() {
        if (b != null) {
            throw new java.lang.ArithmeticException(this + " does not fit in a long");
        }
        return s;
    }

    public BigInteger toBigInteger() {
        return b != null ? b : BigInteger.valueOf(s);
    }

    public int signum() {
        return b != null ? b.signum() : java.lang.Long.signum(s);
    }

    public Int plus(Int o) {
        if (b == null && o.b == null) {
            long r = s + o.s;
            if (((s ^ r) & (o.s ^ r)) >= 0) {
                return of(r);
            }
        }
        return of(toBigInteger().add(o.toBigInteger()));
    }

    public Int minus(Int o) {
        if (b == null && o.b == null) {
            long r = s - o.s;
            if (((s ^ o.s) & (s ^ r)) >= 0) {
                return of(r);
            }
        }
        return of(toBigInteger().subtract(o.toBigInteger()));
    }

    public Int times(Int o) {
        if (b == null && o.b == null) {
            long hi = java.lang.Math.multiplyHigh(s, o.s);
            long lo = s * o.s;
            if (hi == (lo >> 63)) {
                return of(lo);
            }
        }
        return of(toBigInteger().multiply(o.toBigInteger()));
    }

    public Int unaryMinus() {
        if (b == null && s != java.lang.Long.MIN_VALUE) {
            return of(-s);
        }
        return of(toBigInteger().negate());
    }

    public Int abs() {
        return signum() < 0 ? unaryMinus() : this;
    }

    /** Truncating division (semantics 6.3). */
    public Int div(Int o) {
        return div(o, "");
    }

    /** The remainder of truncating division: it has the dividend's sign. */
    public Int rem(Int o) {
        return rem(o, "");
    }

    /** Truncating division; division by zero faults, blamed on {@code symbol}. */
    public Int div(Int o, java.lang.String symbol) {
        if (b == null && o.b == null && o.s != 0 && !(s == java.lang.Long.MIN_VALUE && o.s == -1)) {
            return of(s / o.s);
        }
        if (o.signum() == 0) {
            throw Fault.divisionByZero(symbol);
        }
        return of(toBigInteger().divide(o.toBigInteger()));
    }

    public Int rem(Int o, java.lang.String symbol) {
        if (b == null && o.b == null && o.s != 0 && !(s == java.lang.Long.MIN_VALUE && o.s == -1)) {
            return of(s % o.s);
        }
        if (o.signum() == 0) {
            throw Fault.divisionByZero(symbol);
        }
        return of(toBigInteger().remainder(o.toBigInteger()));
    }

    @Override
    public int compareTo(Int o) {
        if (b == null && o.b == null) {
            return java.lang.Long.compare(s, o.s);
        }
        return toBigInteger().compareTo(o.toBigInteger());
    }

    public boolean lt(Int o) {
        return compareTo(o) < 0;
    }

    public boolean le(Int o) {
        return compareTo(o) <= 0;
    }

    public boolean gt(Int o) {
        return compareTo(o) > 0;
    }

    public boolean ge(Int o) {
        return compareTo(o) >= 0;
    }

    /** The value as a list index or count, or -1 when it is not one. */
    int index() {
        return b == null && s >= 0 && s <= java.lang.Integer.MAX_VALUE ? (int) s : -1;
    }

    @Override
    public boolean equals(java.lang.Object x) {
        if (!(x instanceof Int o)) {
            return false;
        }
        if (b == null) {
            return o.b == null && s == o.s;
        }
        return o.b != null && b.equals(o.b);
    }

    @Override
    public int hashCode() {
        return b != null ? b.hashCode() : java.lang.Long.hashCode(s);
    }

    @Override
    public java.lang.String toString() {
        return b != null ? b.toString() : java.lang.Long.toString(s);
    }

    @Override
    public void writeText(java.lang.StringBuilder sb) {
        sb.append(toString());
    }

    @Override
    public java.lang.Object toJson() {
        return toBigInteger();
    }
}

    private Bytes(byte[] data) {
        this.data = data;
    }

    /** The Bytes of an array, which it copies. */
    public static Bytes of(byte[] data) {
        return new Bytes(data.clone());
    }

    static Bytes wrap(byte[] data) {
        return new Bytes(data);
    }

    byte[] raw() {
        return data;
    }

    /** A copy of the bytes. */
    public byte[] toArray() {
        return data.clone();
    }

    public int length() {
        return data.length;
    }

    public Bytes concat(Bytes o) {
        if (o.data.length == 0) {
            return this;
        }
        if (data.length == 0) {
            return o;
        }
        byte[] r = java.util.Arrays.copyOf(data, data.length + o.data.length);
        java.lang.System.arraycopy(o.data, 0, r, data.length, o.data.length);
        return new Bytes(r);
    }

    java.lang.String hex() {
        java.lang.StringBuilder sb = new java.lang.StringBuilder(data.length * 2);
        for (byte x : data) {
            sb.append(java.lang.Character.forDigit((x >> 4) & 0xF, 16)).append(java.lang.Character.forDigit(x & 0xF, 16));
        }
        return sb.toString();
    }

    @Override
    public boolean equals(java.lang.Object x) {
        return x instanceof Bytes o && java.util.Arrays.equals(data, o.data);
    }

    @Override
    public int hashCode() {
        return java.util.Arrays.hashCode(data);
    }

    @Override
    public java.lang.String toString() {
        return "Bytes(\"" + hex() + "\")";
    }

    @Override
    public void writeText(java.lang.StringBuilder sb) {
        sb.append("Bytes(\"").append(hex()).append("\")");
    }

    @Override
    public java.lang.Object toJson() {
        java.util.Map<java.lang.String, java.lang.Object> m = new java.util.TreeMap<>();
        m.put("$type", "Bytes");
        m.put("hex", hex());
        return m;
    }
}

import java.util.concurrent.atomic.AtomicInteger;

/**
 * A CPL {@code List}: an immutable sequence, a view of a shared array. {@code take}, {@code drop},
 * and {@code drop_last} are views, so they cost nothing; appending to a list that ends where its
 * array's claimed elements end happens in place, so building a list in a loop is linear.
 */
public final class List<T> implements java.lang.Iterable<T>, Value {
    private final java.lang.Object[] a;
    private final int off;
    private final int n;
    /** How many elements of {@code a} are claimed, when lists of it may grow in place. */
    private final AtomicInteger hw;

    private static final List<?> EMPTY = new List<>(new java.lang.Object[0], 0, 0, null);

    private List(java.lang.Object[] a, int off, int n, AtomicInteger hw) {
        this.a = a;
        this.off = off;
        this.n = n;
        this.hw = hw;
    }

    @SuppressWarnings("unchecked")
    public static <T> List<T> empty() {
        return (List<T>) EMPTY;
    }

    /** The list of the given elements, which it copies. */
    @SafeVarargs
    public static <T> List<T> of(T... items) {
        return new List<>(items.clone(), 0, items.length, null);
    }

    /** A list literal of the program. It copies its arguments like {@link #of}: a host that passes
     *  an array it keeps must not be able to change a list CPL holds by changing the array. */
    @SafeVarargs
    public static <T> List<T> lit(T... items) {
        return new List<>(items.clone(), 0, items.length, null);
    }

    public static <T> List<T> copyOf(java.util.Collection<? extends T> items) {
        java.lang.Object[] a = items.toArray();
        return new List<>(a, 0, a.length, null);
    }

    static <T> List<T> wrap(java.lang.Object[] a, int n) {
        return new List<>(a, 0, n, null);
    }

    public int size() {
        return n;
    }

    public boolean isEmpty() {
        return n == 0;
    }

    @SuppressWarnings("unchecked")
    public T get(int i) {
        if (i < 0 || i >= n) {
            throw new java.lang.IndexOutOfBoundsException(i);
        }
        return (T) a[off + i];
    }

    /** The elements {@code from} to {@code to} of this list, sharing its array. */
    public List<T> slice(int from, int to) {
        if (from < 0 || from > to || to > n) {
            throw new java.lang.IndexOutOfBoundsException(from);
        }
        return new List<>(a, off + from, to - from, hw);
    }

    /** Copies the elements into {@code dst} at {@code at}. */
    void copyTo(java.lang.Object[] dst, int at) {
        java.lang.System.arraycopy(a, off, dst, at, n);
    }

    /**
     * A copy with room to grow. It is an {@code Object[]} whatever the array of {@code l} is (a
     * literal keeps its call site's {@code T[]}), so that writing any element into it is safe.
     */
    private static <T> List<T> grown(List<T> l, int extra) {
        java.lang.Object[] b = java.util.Arrays.copyOfRange(l.a, l.off, l.off + java.lang.Math.max(4, 2 * (l.n + extra)), java.lang.Object[].class);
        return new List<>(b, 0, l.n + extra, new AtomicInteger(l.n + extra));
    }

    /** This list with {@code v} appended, in place when no other list has grown its array past it. */
    public List<T> push(T v) {
        int end = off + n;
        if (hw != null && end < a.length && hw.compareAndSet(end, end + 1)) {
            a[end] = v;
            return new List<>(a, off, n + 1, hw);
        }
        List<T> g = grown(this, 1);
        g.a[n] = v;
        return g;
    }

    /** {@code x ++ y}, in place when no other list has grown the array of {@code x} past it. */
    public static <T> List<T> concat(List<T> x, List<T> y) {
        if (y.n == 0) {
            return x;
        }
        if (x.n == 0) {
            return y;
        }
        int end = x.off + x.n;
        if (x.hw != null && end + y.n <= x.a.length && x.hw.compareAndSet(end, end + y.n)) {
            java.lang.System.arraycopy(y.a, y.off, x.a, end, y.n);
            return new List<>(x.a, x.off, x.n + y.n, x.hw);
        }
        List<T> g = grown(x, y.n);
        java.lang.System.arraycopy(y.a, y.off, g.a, x.n, y.n);
        return g;
    }

    /** A copy of the elements as a Java list. */
    @SuppressWarnings("unchecked")
    public java.util.List<T> toJavaList() {
        return (java.util.List<T>) java.util.List.of(java.util.Arrays.copyOfRange(a, off, off + n));
    }

    @Override
    public java.util.Iterator<T> iterator() {
        return new java.util.Iterator<T>() {
            private int i;

            @Override
            public boolean hasNext() {
                return i < n;
            }

            @Override
            @SuppressWarnings("unchecked")
            public T next() {
                if (i >= n) {
                    throw new java.util.NoSuchElementException();
                }
                return (T) a[off + i++];
            }
        };
    }

    @Override
    public boolean equals(java.lang.Object x) {
        if (!(x instanceof List<?> o) || o.n != n) {
            return false;
        }
        for (int i = 0; i < n; i++) {
            if (!a[off + i].equals(o.a[o.off + i])) {
                return false;
            }
        }
        return true;
    }

    @Override
    public int hashCode() {
        int h = 1;
        for (int i = 0; i < n; i++) {
            h = 31 * h + a[off + i].hashCode();
        }
        return h;
    }

    @Override
    public java.lang.String toString() {
        return Rt.valueText(this);
    }

    @Override
    public void writeText(java.lang.StringBuilder sb) {
        sb.append('[');
        for (int i = 0; i < n; i++) {
            if (i > 0) {
                sb.append(", ");
            }
            Rt.writeText(sb, a[off + i]);
        }
        sb.append(']');
    }

    @Override
    public java.lang.Object toJson() {
        java.util.List<java.lang.Object> out = new java.util.ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            out.add(Rt.toJson(a[off + i]));
        }
        return out;
    }
}

/**
 * A fault (semantics 6.7): an execution that cannot continue. It is an error of the program, not
 * a value: a contract that failed, an assertion, a division by zero, a hole, an exhausted resource.
 */
public final class Fault extends java.lang.RuntimeException {
    private static final long serialVersionUID = 1L;

    /** The diagnostic code: contract_violation, assertion_failed, division_by_zero, ... */
    public final java.lang.String code;
    /** For a contract violation: requires, ensures, invariant, or decreases; else null. */
    public final java.lang.String clauseKind;
    /** The failing clause's text (an assertion's expression, a hole's name). */
    public final java.lang.String clause;
    /** The stable identity of the blamed declaration. */
    public final java.lang.String symbol;

    public Fault(java.lang.String code, java.lang.String clauseKind, java.lang.String clause, java.lang.String symbol, java.lang.String message) {
        super(message, null, false, false);
        this.code = code;
        this.clauseKind = clauseKind;
        this.clause = clause;
        this.symbol = symbol;
    }

    static Fault divisionByZero(java.lang.String symbol) {
        return new Fault("division_by_zero", null, "", symbol, "division by zero");
    }

    static Fault requires(java.lang.String clause, java.lang.String symbol) {
        return new Fault("contract_violation", "requires", clause, symbol, "requires clause failed: " + clause);
    }

    /** The fault of an argument that does not satisfy its type (S39). */
    public static Fault refinement(java.lang.String expected, java.lang.String symbol) {
        return new Fault("refinement_violation", null, "", symbol, "the value does not satisfy " + expected);
    }

    /** The fault of an argument that holds null, read where CPL has a value: CPL has no null, so
     *  generated code reads one only when a host passed it (the argument is outside its type). */
    public static Fault nullValue() {
        return new Fault("refinement_violation", null, "", "", "an argument holds null, which is no CPL value");
    }

    /** The fault of a call that exhausted the native stack. */
    public static Fault stack() {
        return new Fault("resource_exhaustion", null, "", "", "the stack is exhausted");
    }

    @Override
    public java.lang.String toString() {
        return code + ": " + getMessage();
    }
}

    private static final Int[] NO_MEASURES = new Int[0];
    private static final java.lang.String[] NO_CLAUSES = new java.lang.String[0];

    private int[] fns = NO_FNS;
    private Int[] measures = NO_MEASURES;
    private int top;

    private java.lang.String[] clauses = NO_CLAUSES;

        this(MAX_CALL_DEPTH);
    }

        this.limit = limit;
    }

    /** A function activation (S38). */
    public void enter() {
        if (++depth > limit) {
            throw new Fault("resource_exhaustion", null, "", "", "the call depth limit of " + limit + " activations was exceeded");
        }
    }

    public void leave() {
        depth--;
    }

    public void measurePush(int f) {
        if (top == fns.length) {
            fns = java.util.Arrays.copyOf(fns, java.lang.Math.max(8, top * 2));
            measures = java.util.Arrays.copyOf(measures, java.lang.Math.max(8, top * 2));
        }
        fns[top] = f;
        measures[top] = null;
        top++;
    }

     * S35: the measure of the current activation of {@code f} is {@code >= 0} and below the measure
     * the nearest enclosing activation of {@code f} recorded, if any.
     */
    public void measureSet(int f, Int m, java.lang.String clause, java.lang.String symbol) {
        for (int i = top - 2; i >= 0; i--) {
            if (fns[i] == f) {
                if (measures[i] != null && m.compareTo(measures[i]) >= 0) {
                    throw new Fault("contract_violation", "decreases", clause, symbol, "decreases clause failed: " + clause);
                }
                break;
            }
        }
        if (m.signum() < 0) {
            throw new Fault("contract_violation", "decreases", clause, symbol, "decreases clause failed: " + clause);
        }
        measures[top - 1] = m;
    }

    public void measurePop() {
        top--;
    }

    public void clauseEnter(java.lang.String owner) {
        if (open == clauses.length) {
            clauses = java.util.Arrays.copyOf(clauses, java.lang.Math.max(8, open * 2));
        }
        clauses[open++] = owner;
    }

    public void clauseLeave() {
        open--;
    }

    /**

     * innermost expression still being evaluated when it arrives is the one it escaped.
     */
    public Fault attribute(Fault f) {
        if (open == 0) {
            return f;
        }
        return new Fault(f.code, f.clauseKind, f.clause, clauses[open - 1], f.getMessage());
    }

    private static void write(java.lang.Object v, java.lang.StringBuilder sb) {
        if (v == null) {
            sb.append("null");
        } else if (v instanceof java.lang.Boolean b) {
            sb.append(b ? "true" : "false");
        } else if (v instanceof java.lang.Number n) {
            sb.append(n.toString());
        } else if (v instanceof java.lang.String s) {
            sb.append('"');
            for (int i = 0; i < s.length(); i++) {
                char c = s.charAt(i);
                if (c == '"') {
                    sb.append("\\\"");
                } else if (c == '\\') {
                    sb.append("\\\\");
                } else if (c == '\n') {
                    sb.append("\\n");
                } else if (c < 0x20 || c > 0x7e) {
                    sb.append(java.lang.String.format("\\u%04x", (int) c));
                } else {
                    sb.append(c);
                }
            }
            sb.append('"');
        } else if (v instanceof java.util.Map<?, ?> m) {
            sb.append('{');
            boolean first = true;
            for (java.util.Map.Entry<?, ?> e : m.entrySet()) {
                if (!first) {
                    sb.append(',');
                }
                first = false;
                write(e.getKey(), sb);
                sb.append(':');
                write(e.getValue(), sb);
            }
            sb.append('}');
        } else if (v instanceof java.util.List<?> l) {
            sb.append('[');
            for (int i = 0; i < l.size(); i++) {
                if (i > 0) {
                    sb.append(',');
                }
                write(l.get(i), sb);
            }
            sb.append(']');
        } else {
            throw new java.lang.IllegalArgumentException("not JSON: " + v.getClass());
        }
    }

    private static final class Parser {
        final java.lang.String s;
        int i;

        Parser(java.lang.String s) {
            this.s = s;
        }

        void ws() {
            while (i < s.length() && " \t\n\r".indexOf(s.charAt(i)) >= 0) {
                i++;
            }
        }

        java.lang.Object value(int depth) {
            if (depth > 512) {
                throw new DecodeError("JSON nested too deeply");
            }
            ws();
            if (i >= s.length()) {
                throw new DecodeError("unexpected end of JSON");
            }
            char c = s.charAt(i);
            if (c == '{') {
                i++;
                java.util.Map<java.lang.String, java.lang.Object> m = new java.util.TreeMap<>();
                ws();
                if (i < s.length() && s.charAt(i) == '}') {
                    i++;
                    return m;
                }
                while (true) {
                    ws();
                    java.lang.String k = string();
                    ws();
                    expect(':');
                    m.put(k, value(depth + 1));
                    ws();
                    if (i < s.length() && s.charAt(i) == ',') {
                        i++;
                    } else {
                        expect('}');
                        return m;
                    }
                }
            }
            if (c == '[') {
                i++;
                java.util.List<java.lang.Object> l = new java.util.ArrayList<>();
                ws();
                if (i < s.length() && s.charAt(i) == ']') {
                    i++;
                    return l;
                }
                while (true) {
                    l.add(value(depth + 1));
                    ws();
                    if (i < s.length() && s.charAt(i) == ',') {
                        i++;
                    } else {
                        expect(']');
                        return l;
                    }
                }
            }
            if (c == '"') {
                return string();
            }
            if (s.startsWith("true", i)) {
                i += 4;
                return java.lang.Boolean.TRUE;
            }
            if (s.startsWith("false", i)) {
                i += 5;
                return java.lang.Boolean.FALSE;
            }
            if (s.startsWith("null", i)) {
                i += 4;
                return null;
            }
            int start = i;
            while (i < s.length() && "+-0123456789.eE".indexOf(s.charAt(i)) >= 0) {
                i++;
            }
            java.lang.String t = s.substring(start, i);
            if (t.isEmpty()) {
                throw new DecodeError("unexpected character at " + start);
            }
            try {
                return new java.math.BigInteger(t);
            } catch (java.lang.NumberFormatException e) {
                return new java.math.BigDecimal(t);
            }
        }

        void expect(char c) {
            if (i >= s.length() || s.charAt(i) != c) {
                throw new DecodeError("expected " + c + " at " + i);
            }
            i++;
        }

        java.lang.String string() {
            expect('"');
            java.lang.StringBuilder sb = new java.lang.StringBuilder();
            while (true) {
                if (i >= s.length()) {
                    throw new DecodeError("unterminated string");
                }
                char c = s.charAt(i++);
                if (c == '"') {
                    return sb.toString();
                }
                if (c != '\\') {
                    sb.append(c);
                    continue;
                }
                char e = s.charAt(i++);
                switch (e) {
                    case '"' -> sb.append('"');
                    case '\\' -> sb.append('\\');
                    case '/' -> sb.append('/');
                    case 'b' -> sb.append('\b');
                    case 'f' -> sb.append('\f');
                    case 'n' -> sb.append('\n');
                    case 'r' -> sb.append('\r');
                    case 't' -> sb.append('\t');
                    case 'u' -> {
                        sb.append((char) java.lang.Integer.parseInt(s.substring(i, i + 4), 16));
                        i += 4;
                    }
                    default -> throw new DecodeError("unknown escape");
                }
            }
        }
    }

        for (int i = 0; i < n; i++) {
            out[n - 1 - i] = l.get(i);
        }
        return List.wrap(out, n);
    }

        for (T x : l) {
            if (x.equals(v)) {
                return true;
            }
        }
        return false;
    }

    public static java.lang.String textTrim(java.lang.String t) {
        return textTrimEnd(textTrimStart(t));
    }

    public static java.lang.String textTrimStart(java.lang.String t) {
        int i = 0;
        while (i < t.length() && isWhiteSpace(t.charAt(i))) {
            i++;
        }
        return i == 0 ? t : t.substring(i);
    }

    public static java.lang.String textTrimEnd(java.lang.String t) {
        int j = t.length();
        while (j > 0 && isWhiteSpace(t.charAt(j - 1))) {
            j--;
        }
        return j == t.length() ? t : t.substring(0, j);
    }

    public static List<java.lang.String> textSplit(java.lang.String t, java.lang.String sep) {
        if (sep.isEmpty()) {
            throw Fault.requires("not sep.is_empty()", "std.text.split");
        }
        java.util.ArrayList<java.lang.Object> out = new java.util.ArrayList<>();
        int from = 0;
        while (true) {
            int at = t.indexOf(sep, from);
            if (at < 0) {
                out.add(t.substring(from));
                break;
            }
            out.add(t.substring(from, at));
            from = at + sep.length();
        }
        return List.wrap(out.toArray(), out.size());
    }

    public static java.lang.String textJoin(List<java.lang.String> parts, java.lang.String sep) {
        java.lang.StringBuilder sb = new java.lang.StringBuilder();
        boolean first = true;
        for (java.lang.String p : parts) {
            if (!first) {
                sb.append(sep);
            }
            first = false;
            sb.append(p);
        }
        return sb.toString();
    }

    public static java.lang.String textToUpper(java.lang.String t) {
        char[] c = null;
        for (int i = 0; i < t.length(); i++) {
            char x = t.charAt(i);
            if (x >= 'a' && x <= 'z') {
                if (c == null) {
                    c = t.toCharArray();
                }
                c[i] = (char) (x - 32);
            }
        }
        return c == null ? t : new java.lang.String(c);
    }

    public static java.lang.String textToLower(java.lang.String t) {
        char[] c = null;
        for (int i = 0; i < t.length(); i++) {
            char x = t.charAt(i);
            if (x >= 'A' && x <= 'Z') {
                if (c == null) {
                    c = t.toCharArray();
                }
                c[i] = (char) (x + 32);
            }
        }
        return c == null ? t : new java.lang.String(c);
    }

    public static Result<Int, ConstraintError> textParseInt(java.lang.String t) {
        Int v = Int.parse(t);
        if (v != null) {
            return Result.ok(v);
        }
        return Result.err(new ConstraintError("Int", quoteText(t) + " is not a decimal integer"));
    }

    public static List<java.lang.String> textChars(java.lang.String t) {
        java.util.ArrayList<java.lang.Object> out = new java.util.ArrayList<>(t.length());
        for (int i = 0; i < t.length(); ) {
            int n = java.lang.Character.charCount(t.codePointAt(i));
            out.add(t.substring(i, i + n));
            i += n;
        }
        return List.wrap(out.toArray(), out.size());
    }

    public static Bytes bytesFromList(List<Int> values) {
        byte[] b = new byte[values.size()];
        int i = 0;
        for (Int v : values) {
            int x = v.index();
            if (x < 0 || x > 255) {
                throw Fault.requires("forall v in values: v >= 0 and v <= 255", "std.bytes.from_list");
            }
            b[i++] = (byte) x;
        }
        return Bytes.wrap(b);
    }

    public static List<Int> bytesToList(Bytes b) {
        byte[] d = b.raw();
        java.lang.Object[] out = new java.lang.Object[d.length];
        for (int i = 0; i < d.length; i++) {
            out[i] = Int.of(d[i] & 0xFF);
        }
        return List.wrap(out, d.length);
    }

    public static Bytes bytesFromText(java.lang.String t) {
        return Bytes.wrap(t.getBytes(java.nio.charset.StandardCharsets.UTF_8));
    }

    /** The offset of the first byte that does not begin a valid UTF-8 sequence, or -1. */
    private static int invalidUtf8(byte[] d) {
        int i = 0;
        while (i < d.length) {
            int c = d[i] & 0xFF;
            int n;
            int min;
            if (c < 0x80) {
                i++;
                continue;
            } else if (c >= 0xC2 && c <= 0xDF) {
                n = 1;
                min = 0x80;
            } else if (c >= 0xE0 && c <= 0xEF) {
                n = 2;
                min = 0x800;
            } else if (c >= 0xF0 && c <= 0xF4) {
                n = 3;
                min = 0x10000;
            } else {
                return i;
            }
            int cp = c & (0x3F >> n);
            for (int k = 1; k <= n; k++) {
                if (i + k >= d.length) {
                    return i;
                }
                int x = d[i + k] & 0xFF;
                if ((x & 0xC0) != 0x80) {
                    return i;
                }
                cp = (cp << 6) | (x & 0x3F);
            }
            if (cp < min || cp > 0x10FFFF || (cp >= 0xD800 && cp <= 0xDFFF)) {
                return i;
            }
            i += n + 1;
        }
        return -1;
    }

    public static Result<java.lang.String, ConstraintError> bytesToText(Bytes b) {
        int bad = invalidUtf8(b.raw());
        if (bad >= 0) {
            return Result.err(new ConstraintError("Text", "invalid UTF-8 at byte " + bad));
        }
        return Result.ok(new java.lang.String(b.raw(), java.nio.charset.StandardCharsets.UTF_8));
    }
